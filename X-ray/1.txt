-- 7.1 Authenticate Patient via Reference Number or Email
IF OBJECT_ID('dentist.usp_AuthenticatePatientPortal', 'P') IS NOT NULL

GO
 
CREATE PROCEDURE [dentist].[usp_AuthenticatePatientPortal]
    @Identifier NVARCHAR(100) -- Can be Reference Number (DEN-2026-XXXXX) OR Email OR Phone
AS
BEGIN
    SET NOCOUNT ON;
    DECLARE @CleanId NVARCHAR(100) = LTRIM(RTRIM(@Identifier));
 
    SELECT TOP 1 
        [PatientID],
        [DoctorID],
        [FirstName],
        [LastName],
        [ReferenceNumber],
        [Email],
        [Phone],
        [DOB],
        [Gender],
        [Region],
        [CurrentTreatmentPlan],
        [TreatmentStage],
        [PasswordHash],
        [IsPortalActive],
        [MustChangePassword],
        [LastLoginAt]
    FROM [dentist].[Patients]
    WHERE ([ReferenceNumber] = @CleanId
       OR LOWER([Email]) = LOWER(@CleanId)
       OR REPLACE(REPLACE(REPLACE(Phone, ' ', ''), '-', ''), '+', '') = REPLACE(REPLACE(REPLACE(@CleanId, ' ', ''), '-', ''), '+', ''))
      AND [IsPortalActive] = 1;
END
GO
 
-- 7.2 Process Payment & Automatically Reconcile Invoice (Card or Cash)
IF OBJECT_ID('dentist.usp_ProcessInvoicePayment', 'P') IS NOT NULL
    DROP PROCEDURE [dentist].[usp_ProcessInvoicePayment];
GO
 
CREATE PROCEDURE [dentist].[usp_ProcessInvoicePayment]
    @InvoiceID            BIGINT,
    @PatientID            INT,
    @Amount               DECIMAL(18,2),
    @PaymentMethod        NVARCHAR(50), -- 'Online_Card', 'Cash', 'Bank_Transfer'
    @TransactionReference NVARCHAR(150) = NULL,
    @PaymentGateway       NVARCHAR(50) = NULL,
    @ReceivedByDoctorID   INT = NULL,
    @Notes                NVARCHAR(MAX) = NULL,
    @GeneratedReceiptNo   NVARCHAR(50) OUTPUT,
    @NewInvoiceStatus     NVARCHAR(30) OUTPUT
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
 
    BEGIN TRANSACTION;
 
    BEGIN TRY
        -- 1. Validate Invoice existence
        DECLARE @CurrentTotal DECIMAL(18,2), @CurrentPaid DECIMAL(18,2), @InvPatientID INT;
        SELECT @CurrentTotal = TotalAmount, @CurrentPaid = PaidAmount, @InvPatientID = PatientID
        FROM [dentist].[Invoices] WITH (UPDLOCK, ROWLOCK)
        WHERE InvoiceID = @InvoiceID;
 
        IF @CurrentTotal IS NULL
        BEGIN
            THROW 50001, 'Invoice not found.', 1;
        END
 
        IF @InvPatientID <> @PatientID
        BEGIN
            THROW 50002, 'Security error: Patient does not own this invoice.', 1;
        END
 
        -- 2. Generate Receipt Number
        DECLARE @Year NVARCHAR(4) = CAST(YEAR(GETDATE()) AS NVARCHAR(4));
        DECLARE @NextId BIGINT = ISNULL((SELECT MAX(PaymentID) FROM [dentist].[Payments]), 0) + 1;
        SET @GeneratedReceiptNo = 'REC-' + @Year + '-' + RIGHT('00000' + CAST(@NextId AS NVARCHAR(10)), 5);
 
        -- 3. Determine Status based on payment method
        DECLARE @PayStatus NVARCHAR(30) = 'Success';
        DECLARE @CashVoucher NVARCHAR(50) = NULL;
 
        IF @PaymentMethod = 'Cash' AND @ReceivedByDoctorID IS NULL
        BEGIN
            -- Patient generated cash payment slip for front desk
            SET @PayStatus = 'Pending_Cash_Verification';
            SET @CashVoucher = 'CSH-' + @Year + '-' + RIGHT('00000' + CAST(@NextId AS NVARCHAR(10)), 5);
        END
 
        -- 4. Record Payment
        INSERT INTO [dentist].[Payments] (
            PaymentReceiptNo,
            InvoiceID,
            PatientID,
            Amount,
            PaymentMethod,
            PaymentStatus,
            TransactionReference,
            PaymentGateway,
            CashVoucherCode,
            ReceivedByDoctorID,
            PaymentDate,
            Notes
        )
        VALUES (
            @GeneratedReceiptNo,
            @InvoiceID,
            @PatientID,
            @Amount,
            @PaymentMethod,
            @PayStatus,
            @TransactionReference,
            @PaymentGateway,
            @CashVoucher,
            @ReceivedByDoctorID,
            SYSUTCDATETIME(),
            @Notes
        );
 
        -- 5. If Payment was successful / confirmed, update invoice ledger
        IF @PayStatus = 'Success'
        BEGIN
            DECLARE @UpdatedPaid DECIMAL(18,2) = @CurrentPaid + @Amount;
            IF @UpdatedPaid >= @CurrentTotal
                SET @NewInvoiceStatus = 'Paid';
            ELSE
                SET @NewInvoiceStatus = 'Partially Paid';
 
            UPDATE [dentist].[Invoices]
            SET PaidAmount = @UpdatedPaid,
                Status = @NewInvoiceStatus,
                UpdatedAt = SYSUTCDATETIME()
            WHERE InvoiceID = @InvoiceID;
        END
        ELSE
        BEGIN
            -- Marked as awaiting cash settlement at reception
            SET @NewInvoiceStatus = 'Pending Cash Settlement';
            UPDATE [dentist].[Invoices]
            SET Status = @NewInvoiceStatus,
                UpdatedAt = SYSUTCDATETIME()
            WHERE InvoiceID = @InvoiceID;
        END
 
        -- 6. Log Clinical Log for Record Keeping
        INSERT INTO [dentist].[ClinicalLogs] (PatientID, DoctorID, Message, LogType, CreatedAt)
        VALUES (
            @PatientID,
            ISNULL(@ReceivedByDoctorID, (SELECT TOP 1 DoctorID FROM [dentist].[Invoices] WHERE InvoiceID = @InvoiceID)),
            'Billing payment recorded: ' + @PaymentMethod + ' of $' + CAST(@Amount AS NVARCHAR(20)) + ' (' + @NewInvoiceStatus + ') Receipt #' + @GeneratedReceiptNo,
            'Billing',
            GETDATE()
        );
 
        COMMIT TRANSACTION;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END
GO
 
-- 7.3 Receptionist / Doctor Confirms Front-Desk Cash Payment
IF OBJECT_ID('dentist.usp_ConfirmCashPaymentAtFrontDesk', 'P') IS NOT NULL
    DROP PROCEDURE [dentist].[usp_ConfirmCashPaymentAtFrontDesk];
GO
 
CREATE PROCEDURE [dentist].[usp_ConfirmCashPaymentAtFrontDesk]
    @CashVoucherCode   NVARCHAR(50),
    @StaffDoctorID     INT,
    @ConfirmedAmount   DECIMAL(18,2) = NULL
AS
BEGIN
    SET NOCOUNT ON;
    SET XACT_ABORT ON;
 
    BEGIN TRANSACTION;
 
    BEGIN TRY
        DECLARE @PaymentID BIGINT, @InvoiceID BIGINT, @PatientID INT, @PaymentAmount DECIMAL(18,2);
        SELECT 
            @PaymentID = PaymentID,
            @InvoiceID = InvoiceID,
            @PatientID = PatientID,
            @PaymentAmount = Amount
        FROM [dentist].[Payments] WITH (UPDLOCK, ROWLOCK)
        WHERE CashVoucherCode = @CashVoucherCode AND PaymentStatus = 'Pending_Cash_Verification';
 
        IF @PaymentID IS NULL
        BEGIN
            THROW 50010, 'Valid pending cash voucher code not found.', 1;
        END
 
        IF @ConfirmedAmount IS NOT NULL AND @ConfirmedAmount > 0
            SET @PaymentAmount = @ConfirmedAmount;
 
        -- Update Payment to Success
        UPDATE [dentist].[Payments]
        SET PaymentStatus = 'Success',
            ReceivedByDoctorID = @StaffDoctorID,
            Amount = @PaymentAmount,
            PaymentDate = SYSUTCDATETIME(),
            Notes = ISNULL(Notes, '') + ' [Cash confirmed at front desk by Staff ID #' + CAST(@StaffDoctorID AS NVARCHAR(10)) + ']'
        WHERE PaymentID = @PaymentID;
 
        -- Reconcile Invoice
        DECLARE @InvTotal DECIMAL(18,2), @InvPaid DECIMAL(18,2);
        SELECT @InvTotal = TotalAmount, @InvPaid = PaidAmount
        FROM [dentist].[Invoices] WITH (UPDLOCK, ROWLOCK)
        WHERE InvoiceID = @InvoiceID;
 
        DECLARE @NewTotalPaid DECIMAL(18,2) = @InvPaid + @PaymentAmount;
        DECLARE @NewStatus NVARCHAR(30) = CASE WHEN @NewTotalPaid >= @InvTotal THEN 'Paid' ELSE 'Partially Paid' END;
 
        UPDATE [dentist].[Invoices]
        SET PaidAmount = @NewTotalPaid,
            Status = @NewStatus,
            UpdatedAt = SYSUTCDATETIME()
        WHERE InvoiceID = @InvoiceID;
 
        -- Audit Log
        INSERT INTO [dentist].[ClinicalLogs] (PatientID, DoctorID, Message, LogType, CreatedAt)
        VALUES (
            @PatientID,
            @StaffDoctorID,
            'Cash Voucher ' + @CashVoucherCode + ' verified and received ($' + CAST(@PaymentAmount AS NVARCHAR(20)) + '). Invoice ' + CAST(@InvoiceID AS NVARCHAR(20)) + ' updated to ' + @NewStatus,
            'Billing',
            GETDATE()
        );
 
        COMMIT TRANSACTION;
 
        SELECT @PaymentID AS ConfirmedPaymentID, @InvoiceID AS InvoiceID, @NewStatus AS InvoiceStatus, @PaymentAmount AS PaidAmount;
    END TRY
    BEGIN CATCH
        IF @@TRANCOUNT > 0 ROLLBACK TRANSACTION;
        THROW;
    END CATCH
END
GO
